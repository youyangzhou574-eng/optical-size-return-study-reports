function Get-D3CloseoutScope($All,$Trusted,[string]$RunRoot,[string]$InputFsp,$OrphanPids) {
    if([string]::IsNullOrWhiteSpace($RunRoot) -or [string]::IsNullOrWhiteSpace($InputFsp)){throw 'D3_CLOSEOUT_EMPTY_SCOPE_REJECTED'}
    if(-not $InputFsp.StartsWith($RunRoot+'\',[StringComparison]::OrdinalIgnoreCase)){throw 'D3_CLOSEOUT_INPUT_OUTSIDE_RUN_REJECTED'}
    $selected=@{};$blocked=@{}
    $runPattern=[regex]::Escape($RunRoot)+'(?=[\\\s"]|$)'
    $inputPattern=[regex]::Escape($InputFsp)+'(?=["\s]|$)'
    foreach($p in $All) {
        $cmd=([string]$p.CommandLine).Replace('/','\')
        $match=([bool]$cmd -and ($cmd -match $runPattern -or $cmd -match $inputPattern))
        $safe=[pscustomobject]@{ProcessId=[int]$p.ProcessId;ParentProcessId=[int]$p.ParentProcessId;Name=[string]$p.Name;CreationDate=[string]$p.CreationDate;CommandAvailable=[bool]$cmd;ExactRunOrInputMatch=$match}
        if($match){$selected[$safe.ProcessId]=$safe}
        foreach($known in $Trusted) {
            if($safe.ProcessId -ne $known.ProcessId){continue}
            $identity=Get-D3RemoteHostState $true $safe.CreationDate ([string]$known.CreationDate)
            if($identity -eq 'SAME_IDENTITY' -and $safe.Name -eq $known.Name){$selected[$safe.ProcessId]=$safe}
            elseif($identity -ne 'PID_REUSED_CONFIRMED'){$blocked[$safe.ProcessId]=$safe}
        }
        if($safe.ProcessId -in $OrphanPids){$blocked[$safe.ProcessId]=$safe}
    }
    do {
        $changed=$false
        foreach($p in $All) {
            $parent=$selected[[int]$p.ParentProcessId]
            if(-not $parent -or $selected.ContainsKey([int]$p.ProcessId)){continue}
            $safe=[pscustomobject]@{ProcessId=[int]$p.ProcessId;ParentProcessId=[int]$p.ParentProcessId;Name=[string]$p.Name;CreationDate=[string]$p.CreationDate;CommandAvailable=([bool]$p.CommandLine);ExactRunOrInputMatch=$false}
            try {
                if([string]::IsNullOrWhiteSpace($safe.CreationDate) -or [string]::IsNullOrWhiteSpace($parent.CreationDate)){throw 'D3_DESCENDANT_BIRTH_UNKNOWN'}
                $childTime=[DateTimeOffset]::Parse($safe.CreationDate,[Globalization.CultureInfo]::InvariantCulture)
                $parentTime=[DateTimeOffset]::Parse($parent.CreationDate,[Globalization.CultureInfo]::InvariantCulture)
                if($childTime -lt $parentTime){throw 'D3_DESCENDANT_PARENT_IDENTITY_CONFLICT'}
                $selected[$safe.ProcessId]=$safe;$changed=$true
            } catch {$blocked[$safe.ProcessId]=$safe}
        }
    } while($changed)
    foreach($p in $All) {
        if($p.Name -in @('fdtd-engine-msmpi.exe','fdtd-solutions.exe','mpiexec.exe','smpd.exe') -and -not $selected.ContainsKey([int]$p.ProcessId)) {
            $blocked[[int]$p.ProcessId]=[pscustomobject]@{ProcessId=[int]$p.ProcessId;ParentProcessId=[int]$p.ParentProcessId;Name=[string]$p.Name;CreationDate=[string]$p.CreationDate;CommandAvailable=([bool]$p.CommandLine);ExactRunOrInputMatch=$false}
        }
    }
    [pscustomobject]@{Passed=($selected.Count -eq 0 -and $blocked.Count -eq 0);Members=@($selected.Values);Blocked=@($blocked.Values);TrustedStartIdentityCount=@($Trusted).Count;ImportedNativeCache=$false;ExternalPidExclusions=@();RunRoot=$RunRoot;InputFsp=$InputFsp}
}

def numeric_mat(path):
    result={}
    with h5py.File(path) as file:
        for name,value in file.items():
            if not isinstance(value,h5py.Dataset) or name.startswith('#'):
                continue
            data=value[...]
            if data.dtype.names:
                data=data['real']+1j*data['imag']
            result[name]=data.transpose(tuple(reversed(range(data.ndim)))).squeeze()
    return result


def validate_raw_data(data):
    required = ['f', 'f_all', 'z', 'index_z_coordinates', 'delta_z', 'Pin', 'area',
                'source_z', 'zR', 'zT', 'zbottom', 'ztop', 'Rraw', 'Traw', 'Pbottom', 'Ptop']
    required += ['nk_' + name for name in ['PDMS', 'ITO', 'Fe', 'water']]
    required += [prefix + axis for prefix in ['E', 'index_'] for axis in ['x', 'y', 'z']]
    required += [f'E{axis}_w{witness}' for witness in [1, 2] for axis in ['x', 'y', 'z']]
    required += ['x', 'y', 'delta_x', 'delta_y', 'index_x_coordinates', 'index_y_coordinates',
                 'index_x_offset', 'index_y_offset', 'index_z_offset']
    required += [name + f'_w{witness}' for witness in [1, 2] for name in ['x', 'y', 'z', 'delta_x', 'delta_y', 'delta_z']]
    for name in required:
        value = np.asarray(data[name])
        if not value.size or not np.all(np.isfinite(value)):
            raise ValueError(f'Empty or non-finite raw data: {name}')
    if np.any(np.asarray(data['f']) <= 0) or np.any(np.asarray(data['Pin']) <= 0) or float(data['area']) <= 0:
        raise ValueError('Frequency, source power and area must be positive')
    if np.asarray(data['f_all']).shape != (91, 8):
        raise ValueError('Expected eight complete 91-point frequency arrays')
    validate_coordinates(data)


def validate_coordinates(data):
    z = np.asarray(data['z']).reshape(-1)
    if len(z) < 2 or np.any(np.diff(z) <= 0):
        raise ValueError('Center Yee z coordinates must ascend')

    def equal(left, right, label):
        a, b = np.asarray(left).reshape(-1), np.asarray(right).reshape(-1)
        if a.shape != b.shape or not np.allclose(a, b, rtol=0, atol=1e-15):
            raise ValueError('Yee coordinate mismatch: ' + label)

    for axis in 'xyz':
        base = np.asarray(data[axis]).reshape(-1)
        delta = np.broadcast_to(np.asarray(data['delta_' + axis]).reshape(-1), base.shape)
        index_base = 'index_z_coordinates' if axis == 'z' else 'index_' + axis + '_coordinates'
        equal(base, data[index_base], axis + ' index base')
        equal(base + delta, data['index_' + axis + '_offset'], axis + ' index offset')
    for witness in [1, 2]:
        observed_z = np.asarray(data[f'z_w{witness}']).reshape(-1)
        dz = np.broadcast_to(np.asarray(data[f'delta_z_w{witness}']).reshape(-1), observed_z.shape)
        equal(observed_z, z, f'witness {witness} base z')
        equal(observed_z + dz, data['index_z_offset'], f'witness {witness} offset z')
        for axis in 'xy':
            if np.asarray(data[f'{axis}_w{witness}']).size != 1:
                raise ValueError('Witness must remain a single actual transverse line')
            equal(data[f'delta_{axis}_w{witness}'], data['delta_' + axis], f'witness {witness} transverse offset {axis}')


def analyze(data):
    validate_raw_data(data)
    f=np.asarray(data['f']).reshape(-1)
    wl=299792458/f*1e9
    if len(f)!=91 or np.max(np.abs(np.sort(wl)-np.arange(350,801,5)))>1e-8:
        raise ValueError('Actual fresh field wavelengths are not 350:5:800 nm')
    if np.max(np.abs(data['f_all']-f[:,None]))>1:
        raise ValueError('Actual field/flux/index frequency pairing failed')
    z=np.asarray(data['z']).reshape(-1)
    if not np.allclose(z,data['index_z_coordinates'],rtol=0,atol=1e-15):
        raise ValueError('Actual field and index Yee origins differ')
    fields={v:np.asarray(data['E'+v]).squeeze() for v in ['x','y','z']}
    index={v:np.asarray(data['index_'+v]).squeeze() for v in ['x','y','z']}
    witness_error=0.
    witness_errors={}
    for witness in [1,2]:
        per_witness_error=0.
        for v in ['x','y','z']:
            observed=np.asarray(data[f'E{v}_w{witness}']).squeeze()
            if observed.shape!=fields[v].shape:
                raise ValueError('Uniformity witness dimensions differ')
            scale=np.max(np.sqrt(sum(np.abs(e)**2 for e in fields.values())),axis=0)
            per_witness_error=max(per_witness_error,float(np.max(np.abs(observed-fields[v])/np.maximum(scale,1e-100))))
        witness_errors[str(witness)]=per_witness_error
        witness_error=max(witness_error,per_witness_error)
    if witness_error>1e-6:
        raise ValueError(f'Flat-plane translational invariance failed: {witness_error}')
    material_nk=np.asarray([np.asarray(data['nk_'+name]).reshape(-1) for name in ['PDMS','ITO','Fe','water']])
    planes={key:float(np.asarray(data[key]).reshape(-1)[0]) for key in ['source_z','zR','zT','zbottom','ztop']}
    area=float(data['area'])
    absorption=component_spectra(f,z,data['delta_z'],fields,index,material_nk**2,data['Pin'],area,planes['zbottom'],planes['ztop'])
    reference=[plane_reference(wl[j],material_nk[:,j],**planes) for j in range(91)]
    incident=np.array([r['control_incident_per_source'] for r in reference])
    kz_imag=2*np.pi*material_nk[3].imag/(wl*1e-9)
    R=data['Rraw']*np.exp(2*kz_imag*(planes['zbottom']-planes['zR']))/incident
    T=data['Traw']*np.exp(2*kz_imag*(planes['zT']-planes['ztop']))/incident
    absorption/=incident[:,None]
    net=(data['Pbottom']-data['Ptop'])/incident
    correction=np.array([r['incident_interference_correction'] for r in reference])
    direct=1-R-T+correction
    closure=net-absorption.sum(axis=1)
    tmm=np.array([[r['R_control'],r['T_control']]+r['material_absorption'] for r in reference])
    actual=np.column_stack([R,T,absorption])
    order=np.argsort(wl)
    x=wl[order]
    solar=np.loadtxt(ROOT/'evidence/a2_preflight_20260930/solar_reference/ASTMG173.csv',delimiter=',',skiprows=2)
    weight=np.interp(x,solar[:,0],solar[:,2])*x
    mean=lambda values:float(np.trapezoid(np.abs(values[order])*weight,x)/np.trapezoid(weight,x))
    errors=np.abs(actual-tmm)
    labels=['R','T','A_PDMS','A_ITO','A_Fe2O3','A_water']
    means={label:mean(errors[:,j]) for j,label in enumerate(labels)}
    maxima={label:float(errors[:,j].max()) for j,label in enumerate(labels)}
    maximum_wavelengths={label:float(wl[np.argmax(errors[:,j])]) for j,label in enumerate(labels)}
    component_pass=mean(closure)<=.005 and np.max(np.abs(closure))<=.02
    flat_pass=all(means[q]<=.002 and maxima[q]<=.01 for q in ['R','T','A_ITO','A_Fe2O3'])
    flux_pass=mean(direct-net)<=.002 and np.max(np.abs(direct-net))<=.005
    summary={'actual_planes_m':planes,'uniformity_max_relative_field_difference':witness_error,
             'uniformity_max_relative_field_difference_per_witness':witness_errors,
             'weighted_abs_component_closure':mean(closure),'max_component_closure':float(np.max(np.abs(closure))),
             'max_component_closure_wavelength_nm':float(wl[np.argmax(np.abs(closure))]),
             'direct_vs_net_weighted_abs_difference':mean(direct-net),'direct_vs_net_max_difference':float(np.max(np.abs(direct-net))),
             'direct_vs_net_max_difference_wavelength_nm':float(wl[np.argmax(np.abs(direct-net))]),
             'flat_TMM_weighted_abs_errors':means,'flat_TMM_max_errors':maxima,
             'flat_TMM_max_error_wavelength_nm':maximum_wavelengths,
             'COMPONENT_PASS':bool(component_pass),'FLAT_CALIBRATION_PASS':bool(flat_pass),'FLUX_REFERENCE_PASS':bool(flux_pass),
             'paired_numerical_convergence':'NOT_YET_TESTED',
             'qualification':'COMPONENT_CALIBRATED_CONVERGENCE_PENDING' if component_pass and flat_pass and flux_pass else 'RAW_RESULT_REQUIRES_REVIEW'}
    table=np.column_stack([wl,actual,tmm,net,direct,closure,1-R-T,correction,1-data['Rraw']-data['Traw'],
                           data['Rraw'],data['Traw'],data['Pbottom'],data['Ptop'],data['Pin'],f])[order]
    if not np.all(np.isfinite(table)):
        raise ValueError('Non-finite derived spectrum cannot be qualified')
    return summary,table,['wavelength_nm']+labels+['TMM_'+v for v in labels]+['A_net','A_direct_corrected','component_closure','one_minus_R_T_control_unadjusted','incident_interference_correction','one_minus_Rraw_Traw_source_planes','R_raw_source_planes','T_raw_source_planes','Pbottom_raw','Ptop_raw','source_power_W','source_frequency_Hz']


def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('raw',type=Path)
    parser.add_argument('--output',type=Path,required=True)
    parser.add_argument('--coordinates',type=Path,required=True)
    args=parser.parse_args()
    if args.output.exists():
        raise ValueError('Refusing to replace an existing result-analysis directory')
    data = numeric_mat(args.raw)
    coordinates = numeric_mat(args.coordinates)
    for name in data.keys() & coordinates.keys():
        if not np.array_equal(data[name], coordinates[name]):
            raise ValueError('Conflicting data between fresh export files: ' + name)
    data.update(coordinates)
    summary,table,columns=analyze(data)
    args.output.mkdir(parents=True)
    np.savetxt(args.output/'F0_X_components_and_calibration_91.csv',table,delimiter=',',header=','.join(columns),comments='',fmt='%.17g')
    (args.output/'F0_X_result_qualification.json').write_text(json.dumps(summary,indent=2)+'\n',encoding='ascii')
    print(json.dumps(summary,indent=2))


if __name__=='__main__':
    main()

"""Flat-stack, component-aligned Yee loss integration; never clips or rescales."""
import numpy as np

EPS0=8.8541878128e-12
BOUNDS=np.array([-10e-6,0,100e-9,125e-9])
# Material order: PDMS, ITO, Fe2O3, water.
REGION_MATERIAL=np.array([3,0,1,2,3])
ADJACENT=[(3,0),(0,1),(1,2),(2,3)]


def loss_parts(effective,materials,component,z,window,tolerance=1e-7):
    effective=np.asarray(effective,dtype=complex).reshape(-1)
    materials=np.asarray(materials,dtype=complex)
    z=np.asarray(z).reshape(-1)
    window=np.asarray(window).reshape(-1)
    if len(effective)!=len(z) or len(window)!=len(z):
        raise ValueError('Yee coordinate shape mismatch')
    if np.min(materials.imag)<-1e-12 or np.min(effective.imag)<-1e-12:
        raise ValueError('Unexplained gain: no absolute-value correction allowed')
    if not np.isfinite(effective).all() or not np.isfinite(materials).all():
        raise ValueError('Nonfinite permittivity')
    result=np.zeros((len(z),4))
    geometric=REGION_MATERIAL[np.searchsorted(BOUNDS,z,side='right')]
    for p,value in enumerate(effective):
        distances=np.abs(value-materials)/np.maximum(1,np.abs(materials))
        near=np.flatnonzero(np.abs(z[p]-BOUNDS)<=window[p])
        allowed={int(geometric[p])}
        for boundary in near:
            allowed.update(ADJACENT[boundary])
        pure=int(np.argmin(distances))
        if distances[pure]<=tolerance and pure in allowed:
            result[p,pure]=value.imag
            continue
        accepted=False
        for boundary in near:
            a,b=ADJACENT[boundary]
            if component=='z':
                observed=1/value
                ma,mb=1/materials[a],1/materials[b]
            else:
                observed=value
                ma,mb=materials[a],materials[b]
            delta=ma-mb
            fraction=float(np.real((observed-mb)*np.conj(delta))/abs(delta)**2)
            reconstructed=fraction*ma+(1-fraction)*mb
            if 0<=fraction<=1 and abs(reconstructed-observed)<=tolerance*max(1,abs(observed)):
                if component=='z':
                    result[p,a]=abs(value)**2*fraction*materials[a].imag/abs(materials[a])**2
                    result[p,b]=abs(value)**2*(1-fraction)*materials[b].imag/abs(materials[b])**2
                else:
                    result[p,a]=fraction*materials[a].imag
                    result[p,b]=(1-fraction)*materials[b].imag
                accepted=True
                break
        if not accepted:
            raise ValueError(f'unassigned Yee permittivity at z={z[p]:.17g}, component={component}, epsilon={value}')
    if np.max(np.abs(result.sum(axis=1)-effective.imag))>tolerance*max(1,float(np.max(np.abs(effective)))):
        raise ValueError('Material decomposition does not reproduce actual loss')
    return result


def integrate_density(z,density,lower,upper):
    z=np.asarray(z).reshape(-1)
    density=np.asarray(density)
    if density.shape[0]!=len(z) or np.any(np.diff(z)<=0):
        raise ValueError('Nonascending density coordinates')
    if not z[0]<=lower<upper<=z[-1]:
        raise ValueError('Control volume not fully captured by raw Yee data')
    inside=(z>lower)&(z<upper)
    coordinates=np.concatenate([[lower],z[inside],[upper]])
    values=np.vstack([[np.interp(lower,z,density[:,i]) for i in range(density.shape[1])],
                      density[inside],
                      [np.interp(upper,z,density[:,i]) for i in range(density.shape[1])]])
    return np.trapezoid(values,coordinates,axis=0)


def component_spectra(f,z,delta_z,fields,index,material_epsilon,pin,area,lower,upper):
    f=np.asarray(f).reshape(-1)
    z=np.asarray(z).reshape(-1)
    dz=np.broadcast_to(np.asarray(delta_z).reshape(-1),z.shape)
    pin=np.asarray(pin).reshape(-1)
    if len(f)!=91 or len(pin)!=91 or np.min(pin)<=0:
        raise ValueError('Invalid frequency samples or incident power')
    output=np.zeros((91,4))
    for component in ['x','y','z']:
        positions=z+dz if component=='z' else z
        window=1.5*np.gradient(positions)
        e=np.squeeze(fields[component])
        eps=np.squeeze(index[component])**2
        if e.shape!=(len(z),91) or eps.shape!=e.shape:
            raise ValueError('Actual index/field shapes do not match: '+component)
        for j,frequency in enumerate(f):
            parts=loss_parts(eps[:,j],material_epsilon[:,j],component,positions,window)
            density=.5*EPS0*2*np.pi*frequency*np.abs(e[:,j,None])**2*parts
            output[j]+=integrate_density(positions,density,lower,upper)*area/pin[j]
    if not np.isfinite(output).all():
        raise ValueError('Nonfinite independent material absorption')
    return output

"""Convert archived MAT/text evidence to verified, standalone TXT/CSV, without solving."""
import argparse
import csv
import hashlib
import io
import json
from pathlib import Path
import shutil
import zipfile
import h5py
import numpy as np


def sha(data):
    return hashlib.sha256(data).hexdigest().upper()


def number(value):
    return str(int(value)) if np.issubdtype(type(value), np.integer) else format(float(value), '.17g')


def numeric_rows(data):
    flat = np.ascontiguousarray(data).reshape(-1)
    if not flat.size:
        return
    # Compare stored bits, so signed zero and close floating values never merge.
    bits = flat.view(np.dtype(('V', flat.dtype.itemsize)))
    starts = np.r_[0, np.flatnonzero(bits[1:] != bits[:-1]) + 1]
    ends = np.r_[starts[1:], flat.size]
    for start, end in zip(starts, ends):
        value = flat[start]
        if flat.dtype.names:
            if flat.dtype.names != ('real', 'imag'):
                raise ValueError('UNSUPPORTED_COMPOUND_TYPE')
            real, imag = number(value['real']), number(value['imag'])
        else:
            real, imag = number(value), ''
        yield dict(start=int(start), count=int(end-start), real=real, imag=imag)


def restore_numeric(rows, shape, dtype):
    flat = np.empty(int(np.prod(shape)), dtype=dtype)
    cursor = 0
    for row in rows:
        start, count = int(row['start']), int(row['count'])
        if start != cursor or count <= 0 or start + count > flat.size:
            raise ValueError('INVALID_RUN_COVERAGE')
        if dtype.names:
            flat['real'][start:start+count] = row['real']
            flat['imag'][start:start+count] = row['imag']
        else:
            flat[start:start+count] = row['real']
        cursor += count
    if cursor != flat.size:
        raise ValueError('INCOMPLETE_RUN_COVERAGE')
    return flat.reshape(shape)


def json_value(value):
    if isinstance(value, (bytes, np.bytes_)):
        return value.decode('utf-8')
    if isinstance(value, np.ndarray):
        return [json_value(v) for v in value.reshape(-1)]
    if isinstance(value, np.generic):
        return json_value(value.item())
    if isinstance(value, (tuple, list)):
        return [json_value(v) for v in value]
    return value


def text_content(data):
    if data.startswith((b'\xff\xfe', b'\xfe\xff')):
        return data.decode('utf-16')
    if b'\x00' in data:
        raise ValueError('UNIDENTIFIED_BINARY_IN_TEXT_ARCHIVE')
    return data.decode('utf-8-sig')


def verified_archive(path):
    archive = zipfile.ZipFile(path)
    manifest = json.loads(archive.read('manifest.json'))
    for item in manifest:
        content = archive.read(item['name'])
        if len(content) != item['bytes'] or sha(content) != item['sha256']:
            raise ValueError('SOURCE_ARCHIVE_HASH_FAILED:'+item['name'])
    return archive


def convert(source, output, *, raw_archive='F0_material_grid_v10_raw_planes.zip',
            evidence_archive='F0_material_grid_v10_evidence.zip', report_version='v10',
            table_name='transverse_material_differences.csv'):
    if output.exists():
        raise ValueError('OUTPUT_EXISTS_NO_OVERWRITE')
    raw_path = source/raw_archive
    evidence_path = source/evidence_archive
    original_hashes = {str(p): sha(p.read_bytes()) for p in
                       [raw_path, evidence_path, source/'REPORT.md', source/table_name]}
    raw = verified_archive(raw_path)
    evidence = verified_archive(evidence_path)
    output.mkdir()
    schema = dict(format='MAT_NUMERIC_RLE_CSV_V1', flatten_order='C in HDF5 storage dimensions',
                  matlab_dimensions='Reverse HDF5 axes; singleton dimensions retained in metadata',
                  run_rule='start is zero-based; repeat real/imag count times; consecutive runs exactly cover each dataset',
                  precision='17 significant decimal digits; verified byte-for-byte numeric reconstruction including signed zero',
                  no_interpolation=True, no_clipping=True, no_solver=True, datasets=[], objects=[])
    raw_csv = output/'material_grid_raw_rle.csv'
    dataset_count = values_count = runs_count = 0
    with raw_csv.open('x', encoding='utf-8', newline='') as handle:
        writer = csv.DictWriter(handle, fieldnames=['source_mat','dataset','start','count','real','imag'])
        writer.writeheader()
        for name in raw.namelist():
            if not name.endswith('.mat'):
                continue
            with h5py.File(io.BytesIO(raw.read(name))) as file:
                def visit(path, item):
                    nonlocal dataset_count, values_count, runs_count
                    obj = dict(source_mat=name, path=path, kind='dataset' if isinstance(item, h5py.Dataset) else 'group',
                               attributes={k: json_value(v) for k,v in item.attrs.items()})
                    schema['objects'].append(obj)
                    if not isinstance(item, h5py.Dataset):
                        return
                    data = item[...]
                    obj['hdf5_shape'] = list(data.shape)
                    if h5py.check_dtype(ref=item.dtype) is not None:
                        obj['reference_targets_C_order'] = [file[ref].name if ref else None for ref in data.reshape(-1)]
                        return
                    if data.dtype.kind not in 'fiuV':
                        raise ValueError('UNSUPPORTED_RAW_DTYPE:'+path)
                    rows = list(numeric_rows(data))
                    if restore_numeric(rows, data.shape, data.dtype).tobytes() != data.tobytes():
                        raise ValueError('NUMERIC_ROUNDTRIP_FAILED:'+path)
                    info = dict(source_mat=name, dataset=path, hdf5_shape=list(data.shape),
                                matlab_shape=list(reversed(data.shape)), dtype=data.dtype.descr if data.dtype.names else data.dtype.str,
                                elements=data.size, runs=len(rows), original_numeric_bytes_sha256=sha(data.tobytes()))
                    if item.attrs.get('MATLAB_class') == b'char':
                        info['decoded_utf16_text'] = np.asarray(data, dtype='<u2').tobytes().decode('utf-16le')
                    schema['datasets'].append(info)
                    dataset_count += 1
                    values_count += data.size
                    runs_count += len(rows)
                    writer.writerows(dict(source_mat=name, dataset=path, **r) for r in rows)
                file.visititems(visit)
    # Read the emitted CSV back, not just the in-memory representation.
    with raw_csv.open(encoding='utf-8', newline='') as handle:
        reader = iter(csv.DictReader(handle))
        current = next(reader, None)
        for info in schema['datasets']:
            rows = []
            while current is not None and (current['source_mat'], current['dataset']) == (info['source_mat'], info['dataset']):
                rows.append(current)
                current = next(reader, None)
            dtype = np.dtype([tuple(v) for v in info['dtype']]) if isinstance(info['dtype'], list) else np.dtype(info['dtype'])
            restored = restore_numeric(rows, tuple(info['hdf5_shape']), dtype)
            if sha(restored.tobytes()) != info['original_numeric_bytes_sha256']:
                raise ValueError('EMITTED_CSV_RECONSTRUCTION_FAILED:'+info['dataset'])
        if current is not None:
            raise ValueError('UNEXPECTED_TRAILING_CSV_ROWS')
    shutil.copy2(source/table_name, output/table_name)
    report = source.joinpath('REPORT.md').read_text(encoding='utf-8')
    prefix = ('WEB-READABLE REPLACEMENT FOR ZIP/MAT ATTACHMENTS\n'
              'Scientific report ID/version remain unchanged; conversion only, no new audit or solve.\n'
              'Upload this report TXT, full_evidence.txt, material_grid_metadata.txt, '
              'material_grid_raw_rle.csv and '+table_name+'.\n'
              'ZIP names inside the original report identify preserved originals, not required web attachments.\n'
              'The raw CSV losslessly represents every numeric dataset, including original dataset references; '
              'metadata lists all dimensions, structures and reference targets. No material/physics claims changed.\n\n')
    (output/'REPORT_FOR_PRO.txt').write_text(prefix+report, encoding='utf-8')
    sections = ['COMPLETE TEXT EVIDENCE FROM PRESERVED '+report_version.upper()+' ZIP; NO SECTIONS OMITTED\n']
    for name in evidence.namelist():
        data = evidence.read(name)
        sections.append('\n===== BEGIN '+name+' | original SHA256 '+sha(data)+' =====\n'+
                        text_content(data)+'\n===== END '+name+' =====\n')
    (output/'full_evidence.txt').write_text(''.join(sections), encoding='utf-8')
    for path, expected in original_hashes.items():
        if sha(Path(path).read_bytes()) != expected:
            raise ValueError('ORIGINAL_CHANGED:'+path)
    receipt = dict(status='CONVERSION_COMPLETE_NOT_WEB_UPLOADED', numeric_datasets=dataset_count,
                   numeric_elements=values_count, csv_runs=runs_count, reconstructed_numeric_bytes_exact=True,
                   all_mat_object_references_exported=True, all_evidence_text_members=len(evidence.namelist()),
                   original_files_unchanged=True, original_file_hashes=original_hashes, new_solver_starts=0,
